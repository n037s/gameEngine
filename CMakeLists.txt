cmake_minimum_required(VERSION 3.21)
project(GameEngine LANGUAGES CXX)

if(NOT MSVC)
    message(FATAL_ERROR "This project requires the MSVC compiler (Visual Studio 2022).")
endif()
if(NOT DEFINED COMPONENT_MANIFEST OR NOT EXISTS "${COMPONENT_MANIFEST}")
    message(FATAL_ERROR "Use bin/bin.bat build generate to generate the component manifest first.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Configurations" FORCE)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
enable_testing()

find_package(SDL3 CONFIG REQUIRED PATHS "${PROJECT_SOURCE_DIR}/extern/SDL3-3.2.26/cmake" NO_DEFAULT_PATH)
find_package(SDL3_ttf CONFIG REQUIRED PATHS "${PROJECT_SOURCE_DIR}/extern/SDL3_ttf-3.1.0/cmake" NO_DEFAULT_PATH)
find_package(SDL3_image CONFIG REQUIRED PATHS "${PROJECT_SOURCE_DIR}/extern/SDL3_image-3.2.4/cmake" NO_DEFAULT_PATH)

add_library(engine_external INTERFACE)
target_link_libraries(engine_external INTERFACE SDL3::SDL3 SDL3_ttf::SDL3_ttf SDL3_image::SDL3_image opengl32)
target_include_directories(engine_external INTERFACE "${PROJECT_SOURCE_DIR}/extern/SDL3_ttf-3.1.0/include/SDL3_ttf")

function(configure_engine_target target)
    target_compile_options(${target} PRIVATE /W3 /permissive-)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
    set_target_properties(${target} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
        VS_DEBUGGER_WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}/src")
endfunction()

function(copy_engine_runtime target)
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:SDL3::SDL3>"
            "$<TARGET_FILE:SDL3_ttf::SDL3_ttf>"
            "$<TARGET_FILE:SDL3_image::SDL3_image>"
            "$<TARGET_FILE_DIR:${target}>"
        VERBATIM)
endfunction()

function(get_engine_sources root output)
    file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${root}/*.cpp" "${root}/*.h" "${root}/*.hpp")
    list(FILTER sources EXCLUDE REGEX "/(test|x64|x86|Win32|Debug|Release|out|CMakeFiles)/")
    if(NOT sources)
        message(FATAL_ERROR "No sources found in ${root}")
    endif()
    set(${output} ${sources} PARENT_SCOPE)
endfunction()

function(configure_engine_component name root)
    configure_engine_target(${name})
    target_link_libraries(${name} PUBLIC engine_external)
    get_target_property(sources ${name} SOURCES)
    set(include_dirs "${root}")
    foreach(source IN LISTS sources)
        get_filename_component(directory "${source}" DIRECTORY)
        list(APPEND include_dirs "${directory}")
    endforeach()
    list(REMOVE_DUPLICATES include_dirs)
    target_include_directories(${name} PUBLIC ${include_dirs})
    source_group(TREE "${root}" FILES ${sources})
endfunction()

file(READ "${COMPONENT_MANIFEST}" components)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${COMPONENT_MANIFEST}")
get_filename_component(manifest_directory "${COMPONENT_MANIFEST}" DIRECTORY)
string(JSON component_count LENGTH "${components}")
math(EXPR last_component "${component_count} - 1")

foreach(index RANGE ${last_component})
    string(JSON name GET "${components}" ${index} name)
    set(component_project "${manifest_directory}/cmake/${name}")
    if(NOT EXISTS "${component_project}/CMakeLists.txt")
        message(FATAL_ERROR "Missing generated CMakeLists.txt for ${name}. Run build generate.")
    endif()
    add_subdirectory("${component_project}" "components/${name}")
endforeach()

foreach(index RANGE ${last_component})
    string(JSON name GET "${components}" ${index} name)
    string(JSON root GET "${components}" ${index} root)
    string(JSON template GET "${components}" ${index} template)
    string(JSON dependency_count LENGTH "${components}" ${index} dependencies)
    if(dependency_count GREATER 0)
        math(EXPR last_dependency "${dependency_count} - 1")
        foreach(dependency_index RANGE ${last_dependency})
            string(JSON dependency GET "${components}" ${index} dependencies ${dependency_index})
            target_link_libraries(${name} PUBLIC ${dependency})
        endforeach()
    endif()
    file(GLOB test_sources CONFIGURE_DEPENDS "${root}/test/*.cpp")
    if(test_sources)
        add_custom_target(${name}_tests)
        set_target_properties(${name}_tests PROPERTIES FOLDER Tests)
        foreach(test_source IN LISTS test_sources)
            get_filename_component(test_name "${test_source}" NAME_WE)
            set(test_target "${name}_${test_name}")
            add_executable(${test_target} EXCLUDE_FROM_ALL "${test_source}")
            configure_engine_target(${test_target})
            if(template STREQUAL "library")
                target_link_libraries(${test_target} PRIVATE ${name})
            else()
                get_target_property(application_sources ${name} SOURCES)
                list(FILTER application_sources EXCLUDE REGEX "/main\\.cpp$")
                get_target_property(application_includes ${name} INCLUDE_DIRECTORIES)
                get_target_property(application_links ${name} LINK_LIBRARIES)
                target_sources(${test_target} PRIVATE ${application_sources})
                target_include_directories(${test_target} PRIVATE ${application_includes})
                target_link_libraries(${test_target} PRIVATE ${application_links})
            endif()
            copy_engine_runtime(${test_target})
            set_target_properties(${test_target} PROPERTIES FOLDER Tests)
            add_dependencies(${name}_tests ${test_target})
            add_test(NAME "${name}.${test_name}" COMMAND $<TARGET_FILE:${test_target}>)
            set_tests_properties("${name}.${test_name}" PROPERTIES
                WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}/src" TIMEOUT 60)
        endforeach()
    endif()
endforeach()

foreach(index RANGE ${last_component})
    string(JSON name GET "${components}" ${index} name)
    string(JSON template GET "${components}" ${index} template)
    if(template STREQUAL "application")
        set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT "${name}")
        break()
    endif()
endforeach()
